+class TestHashSemanticsVectors(unittest.TestCase):
+    """계약의 '값 의미'를 DB 없이 고정 — 32비트 정수화 정의가 md5 표준과 일치하는지 검증 벡터로 확인한다.
+
+    PG 의 (('x' || left(md5(v),8))::bit(32)::bigint) 는 정의상 'md5 앞 8 hex 를 부호없는 32비트 정수로 읽은 값'이다.
+    아래 벡터는 설계 문서(HASH_BUCKET_ORACLE_PORT_ADAPTER_DESIGN §1.1)의 실측 표와 같은 값이며,
+    표현식이 아니라 **계약의 수치 의미**가 바뀌면 여기서 잡힌다(SQL 문자열 비교로는 못 잡는 축).
+    """
+
+    @staticmethod
+    def _h32(canon: str) -> int:
+        import hashlib
+        return int(hashlib.md5(canon.encode("utf-8")).hexdigest()[:8], 16)
+
+    def test_known_vectors(self):
+        for canon, expected in (("S:3:abc", 3276922494), ("S#N", 10499643),
+                                ("D:4:12.5", 584801104),
+                                ("T:26:2026-07-26T00:00:00.000000", 1817468297),
+                                ("hbw1S:3:abc", 976348671)):
+            self.assertEqual(self._h32(canon), expected, canon)
+
+    def test_bucket_is_low_bits_and_in_range(self):
+        """bucket = h32 & (2^k - 1) — 값 범위와 2의 거듭제곱 절단(pow2_floor)이 계약대로인지."""
+        for n_in, n_eff in ((128, 128), (100, 64), (8, 8), (1, 1)):
+            self.assertEqual(hc.pow2_floor(n_in), n_eff)
+            b = self._h32("hbw1S:3:abc") & (n_eff - 1)
+            self.assertTrue(0 <= b < n_eff)
+        self.assertEqual(self._h32("S:3:abc") & 127, 126)          # 설계 §1.1 표와 동일
+        self.assertEqual(self._h32("S#N") & 127, 59)
+
+    def test_bucket_expr_embeds_that_mask(self):
+        """PG 구현체가 실제로 그 마스크(n-1)와 salt 결합을 방출하는지(의미↔표현식 연결 고정)."""
+        pg = PgHashContract()
+        e = pg.bucket_expr("KC", 100, "hbw1")
