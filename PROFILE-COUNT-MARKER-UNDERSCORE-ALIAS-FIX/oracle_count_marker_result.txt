=== PROFILE-COUNT-MARKER-UNDERSCORE-ALIAS-FIX / 오라클 실측 ===
[대상 SQL] CTE(WITH ranked) + LEFT JOIN + window(ROW_NUMBER OVER) 포함 이관 SQL

[파싱] parse_errors=없음  cte_names=['RANKED']
[생성] success=True  method=PARSED_MODEL_CTE_SUBQUERY_COUNT_SQL

── 생성된 원본 COUNT SQL ──────────────────────────────────────────────
SELECT COUNT(*) AS CNT
FROM (
WITH ranked AS (
  SELECT
    x.ID,
    x.C_NUM,
    CASE WHEN t.AMT IS NULL THEN x.C_NUM_PS ELSE ROUND(x.C_NUM_PS + t.AMT, 2) END AS C_NUM_PS,
    x.C_VARCHAR2,
    x.C_CHAR,
    x.C_NVARCHAR2,
    x.C_DATE,
    x.C_TS,
    x.C_TS3,
    x.C_TS6,
    ROW_NUMBER() OVER (PARTITION BY x.C_CHAR ORDER BY x.ID) AS rn
  FROM NXDNP.MV_ORA_XDIFF_SRC x
  LEFT JOIN NXDNP.MV_ORA_TEST_SRC t
    ON MOD(x.ID, 300) + 1 = t.ORDER_ID
)
SELECT
  1 AS mv_count_marker
FROM ranked
WHERE
  rn >= 1
) src_count
