SELECT COUNT(*) AS CNT
FROM (
WITH ranked AS (
  SELECT
    a.sn,
    a.emp_no,
    a.flnm,
    a.orgn_cd,
    CASE WHEN b.sn IS NULL THEN a.sal_amt ELSE COALESCE(b.sal_amt, a.sal_amt) END AS sal_amt,
    a.ymd,
    ROW_NUMBER() OVER (PARTITION BY a.orgn_cd ORDER BY a.sn) AS rn
  FROM ntb.ga_hr_a AS a
  LEFT JOIN ntb.ga_hr_b AS b
    ON a.sn = b.sn
)
SELECT
  1 AS mv_count_marker
FROM ranked
WHERE
  rn >= 1
) src_count

  기준값(ntb.ga_hr_a 전체) = 20000
